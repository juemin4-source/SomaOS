```rust
use std::path::Path;
use std::time::Duration;
use tokio::time::timeout;
use super::context::{WorkspaceContext, ProjectKind, BuildTool, ChangedFile};

/// 主入口：检测工作区并返回 WorkspaceContext。
/// 所有检测都有独立超时，单项失败不阻塞整体。
pub async fn detect_workspace(cwd: &Path) -> WorkspaceContext {
    let mut ctx = WorkspaceContext::new(cwd.to_path_buf());

    // 1. Git 检测
    detect_git(&mut ctx).await;

    // 2. 项目类型检测
    detect_project_kind(&mut ctx).await;

    // 3. 构建工具检测
    detect_build_tools(&mut ctx).await;

    // 4. 权限模式（从环境变量读）
    ctx.permission_mode = detect_permission_mode();

    // 5. 最近会话（通过 task/list RPC — 等 runtime 启动后再做）
    // 见 T4

    ctx
}

/// Git 检测：branch, root, changed_files。超时 5s 降级。
async fn detect_git(ctx: &mut WorkspaceContext) {
    // 先确认是不是 git 仓库
    let git_dir = ctx.root.join(".git");
    if !git_dir.exists() {
        return;  // 非 git 目录，所有 git 字段保持 None
    }

    // branch: git rev-parse --abbrev-ref HEAD
    let branch = run_git(&ctx.root, &["rev-parse", "--abbrev-ref HEAD"]).await;
    ctx.branch = branch;

    // git_root: git rev-parse --show-toplevel
    let root = run_git(&ctx.root, &["rev-parse", "--show-toplevel"]).await;
    ctx.git_root = root.map(PathBuf::from);

    // changed_files: git status --short
    let status_output = run_git(&ctx.root, &["status", "--short"]).await;
    if let Some(output) = status_output {
        ctx.changed_files = output
            .lines()
            .filter_map(|line| {
                if line.len() < 3 { return None; }
                let status = line[..2].trim().to_string();
                let path = line[3..].to_string();
                Some(ChangedFile { path, status })
            })
            .collect();
    }
}

/// 执行 git 命令，超时 5 秒。失败或超时返回 None。
async fn run_git(cwd: &Path, args: &[&str]) -> Option<String> {
    let result = timeout(Duration::from_secs(5), async {
        // 使用与 GitOrgan 相同的路径探测逻辑
        let git_exe = std::env::var("SOMA_GIT_PATH")
            .unwrap_or_else(|_| "git".to_string());
        let output = tokio::process::Command::new(&git_exe)
            .args(args)
            .current_dir(cwd)
            .stdin(std::process::Stdio::null())
            .stdout(std::process::Stdio::piped())
            .stderr(std::process::Stdio::null())
            .output()
            .await
            .ok()?;
        if !output.status.success() { return None; }
        String::from_utf8(output.stdout).ok()
    }).await;

    match result {
        Ok(Some(s)) => Some(s.trim().to_string()),
        _ => None,
    }
}
```
注意：`run_git` 复用了 `resolve_git_exe()` 的路径探测逻辑。但这个函数目前定义在 `soma-capability/src/organ.rs` 中（`fn resolve_git_exe()`），是一个 `OnceLock` 静态缓存。TUI 侧不能直接依赖 `soma-capability` crate（它依赖 tokio process 和重度的能力注册）。因此 detect.rs 中需要自己实现一个简化版的 git 路径查找，或者抽一个共享函数到 `soma-core` 的 utility 模块。

```rust
fn find_git() -> String {
    if let Ok(path) = std::env::var("SOMA_GIT_PATH") {
        if !path.is_empty() { return path; }
```

**项目类型检测：**
```rust
async fn detect_project_kind(ctx: &mut WorkspaceContext) {
    // 检查项目根的关键文件
    let has_cargo = ctx.root.join("Cargo.toml").exists();
    let has_package_json = ctx.root.join("package.json").exists();
    let has_pyproject = ctx.root.join("pyproject.toml").exists()
        || ctx.root.join("requirements.txt").exists();
    let has_go_mod = ctx.root.join("go.mod").exists();

    if has_cargo {
        // 检查是 workspace 还是单个 package
        let content = std::fs::read_to_string(ctx.root.join("Cargo.toml")).ok();
        let is_workspace = content.map_or(false, |c| c.contains("[workspace]"));
        ctx.project_kinds.push(if is_workspace {
            ProjectKind::RustWorkspace
        } else {
            ProjectKind::RustPackage
        });
    }
    if has_package_json { ctx.project_kinds.push(ProjectKind::NodePackage); }
    if has_pyproject { ctx.project_kinds.push(ProjectKind::PythonProject); }
    if has_go_mod { ctx.project_kinds.push(ProjectKind::GoModule); }
    if ctx.project_kinds.is_empty() {
        ctx.project_kinds.push(ProjectKind::Unknown);
    }
}

async fn detect_build_tools(ctx: &mut WorkspaceContext) {
    // 简单版本：检测可执行文件是否存在
    for (name, exe) in &[("cargo", "cargo"), ("node", "node"), ("python", "python3")] {
        if which(exe).is_ok() {
            ctx.build_tools.push(BuildTool {
                name: name.to_string(),
                version: None,
            });
        }
    }
}

fn detect_permission_mode() -> PermissionMode {
    match std::env::var("SOMA_PERMISSION_MODE")
        .unwrap_or_default()
        .to_ascii_lowercase()
        .as_str()
    {
        "read_only" | "readonly" => PermissionMode::ReadOnly,
        "ask_on_write" | "ask" => PermissionMode::AskOnWrite,
        _ => PermissionMode::TrustedWorkspace,
    }
}
```

**`soma-tui/src/lib.rs`** 当前：
```rust
pub fn run() -> Result<(), Box<dyn std::error::Error>> {
    let mut terminal = init_terminal()?;
    let app_result = App::new().run(&mut terminal);
    restore_terminal()?;
    app_result

```

改为：
```rust
pub mod workspace;
pub async fn run() -> Result<(), Box<dyn std::error::Error>> {
    let cwd = std::env::current_dir()?;
    let workspace = workspace::detect::detect_workspace(&cwd).await;

    let mut terminal = init_terminal()?;
    let app = App::new(workspace);
    let app_result = app.run(&mut terminal).await;
    restore_terminal()?;
    app_result
}
```

但注意这个改动会有连锁反应：
1. `App::new()` 需要接受 `WorkspaceContext` 参数 → `App::new(ctx: WorkspaceContext)`
2. `App::run()` 需要从 `fn run(&mut self, terminal)` 变为 `async fn run(&mut self, terminal)`
3. `main.rs`（`soma-cli/src/main.rs`）的 `start_tui()` 需要处理 async
4. 需要处理 tokio runtime 的初始化（当前 TUI 可能没有 tokio runtime）

当前 `soma-cli/src/main.rs` 的 `start_tui()`：
```rust
fn start_tui() -> Result<(), i32> {
    soma_tui::run().map_err(|e| { eprintln!("TUI error: {}", e); 1 })
}
```
改为：
```rust
fn start_tui() -> Result<(), i32> {
    let rt = tokio::runtime::Runtime::new().map_err(|e| {
        eprintln!("Runtime error: {}", e); 1
    })?;
    rt.block_on(soma_tui::run()).map_err(|e| {
        eprintln!("TUI error: {}", e); 1
    })
}
```

## 需要谨慎的地方

1. **`soma_tui::run()` 变为 async** — `main.rs` 的 `start_tui()` 当前是同步的。一定要在 `main.rs` 中创建 tokio runtime 然后用 `block_on` 调用，不能在 `start_tui()` 内部 `tokio::runtime::Runtime::new()` 再 drop——因为 `soma_tui::run()` 内部还会 spawn runtime 子进程。正确的做法是 `main.rs` 里 `rt.block_on(soma_tui::run())` 持有 runtime 直到 TUI 退出。

2. **PermissionMode 类型依赖** — `PermissionMode` 定义在 `soma-core/src/policy/mod.rs` 中。`soma-tui` 的 `Cargo.toml` 需要加 `soma-core` 依赖，或者把 `PermissionMode` 定义移到 `soma-model` 或 `soma-protocol`。如果不改 crate 依赖，可以在 `context.rs` 中定义自己的 `PermissionMode` 枚举，但长期建议移到共享 crate。

3. **`detect_workspace()` 不是阻塞启动的** — 所有检测都是独立的，单项失败不影响整体。启动摘要应该先显示项目名和分支（如果 git 检测已完成），再异步刷新其他信息（changed_files, project_kinds）。但首版可以简单等所有检测完成再渲染。

4. **现有 `soma-tui/src/session.rs`** — 这个文件已经有 Session 序列化逻辑。T4 可能需要复用或替换它。检查一下它的实现，如果它的持久化路径与 Runtime 的 SQLite 冲突，以 Runtime 侧为准。
